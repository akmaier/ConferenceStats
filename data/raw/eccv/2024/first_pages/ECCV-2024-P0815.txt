def train ( model , ref_loader , train_loader , optimizer ) :
for I_r , ( I_s , D_p ) in zip ( ref_loader , train_loader ) :
# Extract features
f_r = Encoderr ( I_r )
F_s = Encoders ( I_s )
# Adjust dimensions
F_r = Convm ( f_r )
# Calculate affinity
A = F . softmax ( F_s @ F_r .T , dim = -1)
# Generate rich - resource depth estimations
D_r = Decoderr ( f_r )
# Compute Prior Depth Fusion
F_o , D_c = PriorDepthFusionModule (A , f_r , D_r )
# Generate depth prediction
D_o = Decoders ( F_o )
# Calculate loss and update weights
# D_p is precomputed pseudo label loaded from dataset
loss = rich_resource_guided_loss ( D_o , D_c , D_p )
optimizer . zero_grad ()
loss . backward ()
optimizer . step ()
